:настройка_дополнительных_параметров_mcc rdf:type comcore:Process ;
    rdfs:label "Настройка дополнительных параметров MCC"@ru ;
    comcore:hasStatus :статус_настройки ;
    comcore:isPartOf :процесс_управления_mcc.

:статус_настройки rdf:type comcore:Status ;
    rdfs:label "Статус настройки"@ru ;
    comcore:isResponsibleFor :настройка_дополнительных_параметров_mcc.

:процесс_управления_mcc rdf:type comcore:Process ;
    rdfs:label "Процесс управления MCC"@ru ;
    comcore:hasPart :настройка_дополнительных_параметров_mcc ;
    comcore:isActorOf :оператор_mcc.

:оператор_mcc rdf:type comcore:Agent ;
    rdfs:label "Оператор MCC"@ru ;
    comcore:isResponsibleFor :настройка_дополнительных_параметров_mcc ;
    comcore:Initiates :настройка_дополнительных_параметров_mcc.

:параметры_mcc rdf:type comcore:Resource ;
    rdfs:label "Параметры MCC"@ru ;
    comcore:isResourceOf :настройка_дополнительных_параметров_mcc ;
    comcore:isPartOf :процесс_управления_mcc.
